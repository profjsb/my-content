{{- /* Author list for a publication. Context: dict "authors" "show" "threshold" "link" . Lists longer than threshold show the first `show` names (plus J. S. Bloom when that name comes later) and expand in place via
; link turns names into links to their author pages. */ -}} {{- $authors := .authors -}} {{- $n := len $authors -}} {{- $link := .link | default false -}} {{- if gt $n 0 -}} {{- $me := -1 -}} {{- range $i, $a := $authors }}{{ if and (lt $me 0) (findRE `^(J\.|Joshua)( S\.| Simon)? Bloom$` $a) }}{{ $me = $i }}{{ end }}{{ end -}} {{- $full := slice -}} {{- range $authors -}} {{- $name := partial "pub_author.html" . -}} {{- if $link }}{{ with site.GetPage (printf "/authors/%s" (. | anchorize)) }}{{ $name = printf "%s" .RelPermalink $name }}{{ end }}{{ end -}} {{- $full = $full | append $name -}} {{- end -}} {{- if le $n .threshold }}
{{ delimit $full ", " | safeHTML }}
{{- else }} {{- $short := delimit (first .show $full) ", " -}} {{- $shown := .show -}} {{- if ge $me .show }} {{- $short = printf "%s … %s …" $short (index $full $me) -}} {{- $shown = add $shown 1 -}} {{- else }} {{- $short = printf "%s …" $short -}} {{- end }}
{{ $short | safeHTML }} +{{ sub $n $shown }} moreShow fewer
{{ delimit $full ", " | safeHTML }}
{{- end }} {{- end -}}
