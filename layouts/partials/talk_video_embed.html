{{- /* Embed a watch frame when url_video is a YouTube or Vimeo link; other hosts get no embed (the Video button still links out). A t=/start= seconds param on YouTube links (e.g. ?t=1635s) carries into the embed, so talks buried in long livestreams start at the right offset. */ -}} {{- $v := .Params.url_video | default "" -}} {{- if $v -}} {{- $yt := "" -}} {{- if findRE `youtube\.com/watch` $v -}} {{- $yt = replaceRE `^.*[?&]v=([A-Za-z0-9_-]+).*$` "$1" $v -}} {{- else if findRE `youtube\.com/(live|embed|shorts)/` $v -}} {{- $yt = replaceRE `^.*youtube\.com/(?:live|embed|shorts)/([A-Za-z0-9_-]+).*$` "$1" $v -}} {{- else if findRE `youtu\.be/` $v -}} {{- $yt = replaceRE `^.*youtu\.be/([A-Za-z0-9_-]+).*$` "$1" $v -}} {{- end -}} {{- if and $yt (ne $yt $v) -}} {{- $start := "" -}} {{- if findRE `[?&](t|start)=\d+` $v -}} {{- $start = replaceRE `^.*[?&](?:t|start)=(\d+).*$` "$1" $v -}} {{- end -}}
{{- else if findRE `vimeo\.com/\d+` $v -}} {{- $vm := replaceRE `^.*vimeo\.com/(\d+).*$` "$1" $v -}} {{- if ne $vm $v -}}
{{- end -}} {{- end -}} {{- end -}}
